long long n,a[maxn],b[maxn],cnt,c[maxn];

void add(long long x,long long val)

	for(long long i=x;i<=n;i+=lowbit(i))

long long query(long long x)
{
	long long sum=0;
	for(long long i=x;i;i-=lowbit(i))
		sum+=t1[i]*(x+1)-t2[i];
	return sum;
